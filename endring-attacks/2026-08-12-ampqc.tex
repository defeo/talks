      \draw (E1) edge[bend left] node[below]{$π$} (E);
    }
  \end{tikzpicture}

  \bigskip
  
  \begin{itemize}
    \setlength{\itemsep}{1em}
  \item $O(n)$ isogenies of degree $n$ from $E$
  \item $⇒$ $O(n^2)$ isogenies of degree $≤ n$ from $E$
  \item<2-> $⇒$ If we're lucky, we can meet-in-the-middle at cost
    $O(p^{1/3})$.
  \end{itemize}
\end{frame}
